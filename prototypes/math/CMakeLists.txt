cmake_minimum_required(VERSION 3.16)
project(ink_math_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/.deps" CACHE PATH "Pinned source cache")
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(tinyxml2_BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(${DEPS}/freetype freetype)
target_include_directories(freetype PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(freetype PRIVATE FT_CONFIG_MODULES_H="ink_ft_modules.h")
add_subdirectory(${DEPS}/tinyxml2 tinyxml2)
# Deliberately omit desktop platform adapters, samples and latex.cpp's resource search.
set(MT ${DEPS}/MicroTeX/src)
file(GLOB MT_CORE CONFIGURE_DEPENDS ${MT}/atom/*.cpp ${MT}/box/*.cpp
  ${MT}/core/*.cpp ${MT}/fonts/*.cpp ${MT}/utils/*.cpp
  ${MT}/res/builtin/*.cpp ${MT}/res/font/*.cpp ${MT}/res/parser/*.cpp
  ${MT}/res/reg/*.cpp ${MT}/res/sym/*.cpp)
add_library(microtex STATIC ${MT_CORE} ${MT}/render.cpp)
target_include_directories(microtex SYSTEM PUBLIC ${MT})
target_link_libraries(microtex PRIVATE tinyxml2)
add_library(md4c STATIC ${DEPS}/md4c/src/md4c.c)
target_include_directories(md4c PUBLIC ${DEPS}/md4c/src)
add_executable(math-probe probe.cpp ink_math.cpp)
target_compile_options(math-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(math-probe PRIVATE microtex freetype md4c)
